      <p class='mkup_p'>Back in the early days of PC's I got a BASIC version of Eliza (on a Tarbell audio cassette if I remember correctly) and was fascinated by it. For a few years I have wanted a Tcl version, and <a class='mkup_a mkup_known' href='/page/A+simple+version+of+Eliza'>A simple version of Eliza</a> wasn't really close enough.</p><p class='mkup_p'>The feature that made the original seem so human to me was that any of several key words or phrases would trigger a category, and in the category there would be several possible responses, some including part of the input.</p><p class='mkup_p'>Here is my implementation of the logic, including an early set of dialog data. The first category was added by me. My other favorite oracle was a Magic 8-Ball, so that is now the first in the set. Note that each category of key phrases to be recognized is terminated with an exclamation point by itself in a line, and each category's responses are terminated similarly but with a period. That distinction is purely historical, since my software does not distinguish between the two terminations, and just assumes that groups of key phrases and responses alternate.</p><p class='mkup_p'>The script sets up a window for the dialog, but the makedict and response procs could be used by themselves without Tk. </p><div class='sh_sourceCode'><button class='copybtn btn pull-right' data-clipboard-target='#mkup_code_0' title='Click to copy code snippet to clipboard'><span class='glyphicon glyphicon-copy' aria-hidden='true'></span></button><pre id='mkup_code_0' class='sh_tcl sh_sourceCode'>package require Tcl 8.5
package require Tk
namespace eval ::Eliza {

proc makedict {input} {
    variable dict {}
    set phraselist [split [string map {\n.\n \uffff \n!\n \uffff} $input] \uffff]
    set index 0
# Kick off the loop by grabbing the first keywords
    set phraselist [lassign $phraselist keywords responses] 
    while {$keywords ne &quot;&quot;} {
        dict set dict keywords $index [split $keywords \n] 
        dict set dict responses $index [split $responses \n]
        incr  index
        set phraselist [lassign $phraselist keywords responses] 
    }
}

proc dialog {} {
    if {[winfo exists .eliza]} {destroy .eliza}
    toplevel .eliza
    label .eliza.l -text &quot;Eliza will see you now. Please state your problem.&quot;
    entry .eliza.e -width 80
    frame .eliza.f
    scrollbar .eliza.sb -orient vert -command {.eliza.t yview}
    text .eliza.t -wrap word -yscrollcommand {.eliza.sb set} -padx 5
    pack .eliza.sb  -in .eliza.f -side right -fill y 
    pack  .eliza.t -in .eliza.f -side right -fill both -expand yes 
    pack .eliza.l .eliza.e -padx 5 -fill both
    pack .eliza.f .eliza.t -pady 5 -expand yes -fill both
    focus .eliza.e
    bind .eliza.e &lt;Return&gt; ::Eliza::process
}

proc process {} {
    set input [.eliza.e get]
    .eliza.t insert end &quot;Q. $input\nA. [response $input]\n\n&quot;
    .eliza.t see end
    .eliza.e delete 0 end
}

proc response input {
    variable dict
    set input [string toupper $input]
    foreach {index keyphraselist} [dict get $dict keywords] {
        foreach keyphrase $keyphraselist {
            if {[regexp &quot;\\y$keyphrase\\y(.*)&quot; $input -&gt; remainder]} {
                set possibleresponses [dict get $dict responses $index]
                set which [expr {int ([llength $possibleresponses] * rand())}]
                set response [lindex $possibleresponses $which]
                if {[string index $response end] eq &quot;*&quot;} {
                    return &quot;[string range $response 0 end-1]$remainder?&quot;
                } else {
                    return $response
                }
                
            }
        }
    }
    set index [expr {[llength [dict get $dict keywords]] / 2 - 1}]
    set possibleresponses [dict get $dict responses $index]
    set which [expr {int ([llength $possibleresponses] * rand())}]
    return [lindex $possibleresponses $which]
}
}
